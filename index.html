function doPost(e) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const sheetName = e.parameter.sheetName; const name = e.parameter.name; const totalDays = Number(e.parameter.totalDays); const totalAmount = Number(e.parameter.totalAmount); const dates = e.parameter.dates; const spreadsheet = SpreadsheetApp.getActiveSpreadsheet(); const sheet = spreadsheet.getSheetByName(sheetName); if (!sheet) { throw new Error("找不到工作表：" + sheetName); } // 尋找這位員工是否已經登記 const lastRow = sheet.getLastRow(); let targetRow = 0; if (lastRow >= 2) { const names = sheet .getRange(2, 2, lastRow - 1, 1) .getValues(); for (let i = 0; i < names.length; i++) { if (names[i][0] === name) { targetRow = i + 2; break; } } } const rowData = [ new Date(), name, totalDays, totalAmount, dates ]; // 已登記就更新，尚未登記就新增 if (targetRow > 0) { sheet.getRange(targetRow, 1, 1, 5).setValues([rowData]); } else { sheet.appendRow(rowData); } return ContentService .createTextOutput("success") .setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService .createTextOutput("error: " + error.message) .setMimeType(ContentService.MimeType.TEXT); } finally { lock.releaseLock(); } }
